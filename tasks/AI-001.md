**Phase**: 2
**Effort**: 6-8 hours
**Deps**: F-005, F-010

## Goal

Single chokepoint for all LLM calls. Implements task config, model fallback chain, Redis cache, Anthropic prompt cache, cost tracking, Langfuse tracing.

## Acceptance criteria

- [ ] `packages/ai/router.ts` exports `router` singleton
- [ ] `router.complete(input)` typed end-to-end via task registry
- [ ] Task registry in `packages/ai/tasks.ts` with at least 5 tasks defined
- [ ] Model registry with cost per token
- [ ] Fallback chain executes on primary failure
- [ ] Budget check + enforcement (soft warn, hard limit)
- [ ] L1 (LRU memory) + L2 (Redis) cache
- [ ] Anthropic prompt cache used for system prompts > 2k tokens
- [ ] Every call writes ai_calls row + Langfuse trace
- [ ] Streaming support via AsyncIterable
- [ ] Lint rule blocks direct imports of `@anthropic-ai/sdk` or `openai` outside router
- [ ] Tests: each fallback path, budget enforcement, cache hit, streaming

### Pipeline implementation

```typescript
class Router {
  async complete<T extends keyof Tasks>(req: CompleteInput<T>): Promise<RouterResult<T>> {
    const traceId = newTraceId();
    const trace = this.tracing.startTrace({ traceId, ...req });

    try {
      // 1. Resolve task
      const task = tasks[req.task];
      if (!task) throw new RouterError('UNKNOWN_TASK', `Task ${req.task} not registered`);

      // 2. Budget check
      if (!task.bypassBudget) {
        await this.budget.checkOrThrow(req.userId);
      }

      // 3. Cache lookup
      if (task.cacheable && req.options?.cacheKey !== false) {
        const cached = await this.cache.get(this.cacheKey(req, task));
        if (cached) {
          trace.markCached();
          return { ...cached, cached: true, traceId };
        }
      }

      // 4. Try primary, then fallbacks
      const chain = [task.primaryModel, ...(task.fallback ?? [])];
      let lastError: Error | undefined;

      for (const modelId of chain) {
        try {
          const result = await this.callModel(modelId, task, req, trace);
          // 5. Track + cache
          await this.recordCall({ ...result, userId: req.userId, agent: req.agent, task: req.task });
          await this.budget.charge(req.userId, result.costCents);
          if (task.cacheable) {
            await this.cache.set(this.cacheKey(req, task), result, task.cacheTtlSeconds ?? 3600);
          }
          return { ...result, cached: false, traceId };
        } catch (err) {
          lastError = err as Error;
          trace.recordFallback(modelId, err);
        }
      }

      throw lastError ?? new RouterError('ALL_MODELS_FAILED');
    } finally {
      trace.end();
    }
  }

  private async callModel(
    modelId: ModelId,
    task: Task,
    req: CompleteInput,
    trace: Trace,
  ): Promise<ModelResult> {
    const provider = this.providers[modelInfo(modelId).provider];
    const prompt = await this.loadPrompt(task.promptPath, req.promptVersion);
    const messages = await this.buildMessages(prompt, req.input);

    const start = performance.now();
    const response = await provider.complete({
      model: modelId,
      messages,
      maxTokens: task.maxTokens,
      temperature: task.temperature,
      schema: task.outputSchema,
      stream: req.options?.stream ?? false,
      promptCache: prompt.systemTokens > 2000,
    });
    const latencyMs = performance.now() - start;

    return {
      output: response.output,
      model: modelId,
      tokensIn: response.usage.input,
      tokensOut: response.usage.output,
      costCents: computeCost(modelId, response.usage),
      latencyMs,
    };
  }
}

export const router = new Router(/* deps */);
```

### Budget

```typescript
export class Budget {
  async checkOrThrow(userId: string): Promise<void> {
    const today = new Date().toISOString().slice(0, 10);
    const row = await db
      .select()
      .from(schema.userDailyBudget)
      .where(and(eq(schema.userDailyBudget.userId, userId), eq(schema.userDailyBudget.date, today)));

    const budget = row[0]?.budgetCents ?? env.DEFAULT_USER_DAILY_BUDGET_CENTS;
    const spent = row[0]?.spentCents ?? 0;
    const hardLimit = budget * env.HARD_LIMIT_MULTIPLIER;

    if (spent >= hardLimit) {
      throw new RouterError('BUDGET_HARD_LIMIT', `User ${userId} exceeded hard limit`);
    }
    // soft warning at 70%
    if (spent >= budget * 0.7 && !row[0]?.warningsSent?.['70']) {
      await this.sendWarning(userId, '70');
    }
  }

  async charge(userId: string, cents: number): Promise<void> {
    const today = new Date().toISOString().slice(0, 10);
    await db.execute(sql`
      INSERT INTO user_daily_budget (user_id, date, budget_cents, spent_cents)
      VALUES (${userId}, ${today}, ${env.DEFAULT_USER_DAILY_BUDGET_CENTS}, ${cents})
      ON CONFLICT (user_id, date)
      DO UPDATE SET spent_cents = user_daily_budget.spent_cents + ${cents};
    `);
  }
}
```

## Notes

- Store prompts as markdown files; loaded + interpolated at runtime
- Idempotency: same `req` (modulo non-deterministic options) returns same cached result
- Streaming: cost computed at end of stream
- Never log raw prompt content to Sentry/PostHog (PII risk); send to Langfuse only
