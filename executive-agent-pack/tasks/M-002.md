### Core algorithm

```typescript
async recall(query: string, options: RecallOptions = {}): Promise<RecalledMemory[]> {
  const limit = options.limit ?? 20;
  const minConfidence = options.minConfidence ?? 30;

  // 1. Cache check
  const cacheKey = this.buildCacheKey(query, options);
  const cached = await this.cache.get(cacheKey);
  if (cached) return cached;

  // 2. Embed query
  const embedding = await this.embedQuery(query);

  // 3. Semantic search (overfetch for re-ranking)
  const semantic = await this.semanticSearch(embedding, {
    limit: limit * 3,
    minConfidence,
    types: options.types,
    timeWindow: options.timeWindow,
  });

  // 4. Entity expansion
  const mentionedEntityIds = await this.entityExtractor.extract(query, this.userId);
  const graphExpanded = mentionedEntityIds.length > 0
    ? await this.getMemoriesForEntities(mentionedEntityIds, options)
    : [];

  // 5. Merge + dedup
  const merged = this.mergeUnique([...semantic, ...graphExpanded]);

  // 6. Apply entity filter
  const filtered = options.entityFilter
    ? merged.filter(m => m.memory.entities.some(e => options.entityFilter!.includes(e)))
    : merged;

  // 7. Re-rank
  const ranked = this.reranker.rank(filtered, query);

  // 8. Top N
  const top = ranked.slice(0, limit);

  // 9. Cache
  await this.cache.set(cacheKey, top, 60);

  return top;
}
```

### Re-ranker

```typescript
export function rankMemories(
  candidates: ScoredMemory[],
  weights = { semantic: 0.55, recency: 0.20, confidence: 0.15, importance: 0.10 }
): ScoredMemory[] {
  const now = Date.now();
  return candidates
    .map(c => ({
      ...c,
      score:
        weights.semantic * (c.semanticSim ?? 0) +
        weights.recency * recencyScore(c.memory.createdAt, now) +
        weights.confidence * (c.memory.confidence / 100) +
        weights.importance * (c.memory.importance ?? 0) / 100,
    }))
    .sort((a, b) => b.score - a.score);
}

function recencyScore(date: Date, now: number): number {
  const ageDays = (now - date.getTime()) / (1000 * 60 * 60 * 24);
  return Math.exp(-ageDays / 90);  // half-life 90 days
}
```

```typescript
import { sql } from 'drizzle-orm';

export const semanticMemorySearch = sql<MemoryRow>`
  SELECT m.*, 1 - (m.embedding <=> ${embedding}::vector) AS similarity
  FROM memories m
  WHERE m.user_id = ${userId}
    AND m.confidence >= ${minConfidence}
    ${types ? sql`AND m.type = ANY(${types})` : sql``}
  ORDER BY m.embedding <=> ${embedding}::vector
  LIMIT ${limit}
`;
```

### Entity extractor (initial naive impl)

```typescript
export class EntityExtractor {
  constructor(private userId: string, private db: Database) {}

  async extract(text: string): Promise<string[]> {
    const entities = await this.db
      .select()
      .from(schema.entities)
      .where(eq(schema.entities.userId, this.userId));

    const matched: string[] = [];
    const lowered = text.toLowerCase();
    for (const e of entities) {
      // Naive: token boundary match on name
      const namePattern = new RegExp(`\\b${escapeRegex(e.name)}\\b`, 'i');
      if (namePattern.test(lowered)) matched.push(e.id);
    }
    return matched;
  }
}
```

```typescript
private buildCacheKey(query: string, options: RecallOptions): string {
```
